cmake_minimum_required(VERSION 3.20)
project(LegionNativeControls LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
add_executable(assertions value.c branch.cc runtime.cxx main.cpp)
target_include_directories(assertions PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(assertions PRIVATE -Wall -Wextra)
enable_testing()
add_test(NAME four_assertions COMMAND assertions)
set_tests_properties(four_assertions PROPERTIES LABELS "legion-assertions=4")
